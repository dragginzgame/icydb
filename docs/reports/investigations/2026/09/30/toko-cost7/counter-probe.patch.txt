+fn consumed_cycles(fixture: &ManagedComponentGroupFixture, shard: Principal) -> u128 {
+    let reply = fixture
+        .pic()
+        .query_call_with_effective_principal(
+            Principal::management_canister(),
+            serde_json::from_value(json!("None")).expect("effective principal"),
+            fixture.pic().get_controllers(shard)[0],
+            "canister_metrics",
+            candid::encode_one(CycleProbeArgs { canister_id: shard }).expect("metrics args"),
+        )
+        .expect("read-only replica metrics");
+    let metrics: CycleProbeResult = candid::decode_one(&reply).expect("metrics result");
+    let c = metrics.cycles_consumed;
+    [
+        c.memory,
+        c.compute_allocation,
+        c.ingress_induction,
+        c.instructions,
+        c.request_and_response_transmission,
+        c.canister_creation,
+        c.http_outcalls,
+        c.burned_cycles,
+        c.uninstall,
+    ]
+    .into_iter()
+    .map(|n| u128::try_from(n.0).expect("cycle counter fits u128"))
+    .try_fold(0u128, u128::checked_add)
+    .expect("cycle sum fits u128")
