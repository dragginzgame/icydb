//! Temporary paired IC instruction probe for selected authority reuse.

use icydb_testing_integration::install_prebuilt_fixture_canister;

#[test]
fn selected_authority_preserves_reports_and_reduces_instructions() {
    let directory = std::env::var("ICYDB_SELECTED_AUTHORITY_DIR").unwrap();
    let load = |name| std::fs::read(format!("{directory}/{name}.wasm")).unwrap();
    let before = install_prebuilt_fixture_canister("one_entity_typed_query", load("before"));
    let after = install_prebuilt_fixture_canister("one_entity_typed_query", load("after"));
    type Samples = Result<Vec<(u64, u64, String, u64)>, u16>;
    for kind in 0..3_u8 {
        let baseline: Samples = before
            .query_candid("measure_typed_explain", (kind,))
            .unwrap();
        let candidate: Samples = after
            .query_candid("measure_typed_explain", (kind,))
            .unwrap();
        let baseline = baseline.unwrap();
        let candidate = candidate.unwrap();
        assert_eq!(baseline.len(), 3);
        assert_eq!(candidate.len(), 3);
        for (call, (before, after)) in baseline.iter().zip(&candidate).enumerate() {
            assert_eq!(before.2, after.2);
            assert!(after.0 < before.0);
            assert!(after.0 + after.1 < before.0 + before.1);
            println!(
                "kind={kind} call={call} before_plan={} after_plan={} before_binding={} after_binding={} before_render={} after_render={}",
                before.0, after.0, before.3, after.3, before.1, after.1
            );
        }
    }
}
